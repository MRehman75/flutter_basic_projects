import 'package:flutter/material.dart';

void main() {
  runApp(FlashcardApp());
}

class FlashcardApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flashcards',
      debugShowCheckedModeBanner: false,
      theme: ThemeData.dark(),
      home: HomeScreen(),
    );
  }
}

class HomeScreen extends StatelessWidget {
  final List<String> categories = ["Flutter", "Dart", "Networking"];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text("Select Category")),
      body: Padding(
        padding: const EdgeInsets.all(12.0),
        child: GridView.builder(
          itemCount: categories.length,
          gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
            crossAxisCount: 2,
            childAspectRatio: 1.2,
            crossAxisSpacing: 10,
            mainAxisSpacing: 10,
          ),
          itemBuilder: (context, index) {
            return GestureDetector(
              onTap: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(builder: (context) => FlashcardScreen(category: categories[index])),
                );
              },
              child: Card(
                color: Colors.deepPurpleAccent,
                shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(15)),
                elevation: 5,
                child: Center(
                  child: Text(
                    categories[index],
                    style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold, color: Colors.white),
                  ),
                ),
              ),
            );
          },
        ),
      ),
    );
  }
}

class FlashcardScreen extends StatefulWidget {
  final String category;
  FlashcardScreen({required this.category});

  @override
  _FlashcardScreenState createState() => _FlashcardScreenState();
}

class _FlashcardScreenState extends State<FlashcardScreen> {
  List<Map<String, String>> flashcards = [];
  int currentIndex = 0;
  int score = 0;

  @override
  void initState() {
    super.initState();
    loadFlashcards();
  }

  void loadFlashcards() {
    if (widget.category == "Flutter") {
      flashcards = [
        {"question": "What is Flutter?", "answer": "A UI toolkit by Google."},
        {"question": "What is a Stateful Widget?", "answer": "A widget that maintains state."},
        {"question": "What is Hot Reload?", "answer": "A feature for fast development."},
      ];
    } else if (widget.category == "Dart") {
      flashcards = [
        {"question": "What is Dart?", "answer": "A language for Flutter development."},
        {"question": "What is Null Safety?", "answer": "Prevents null reference errors."},
        {"question": "What is an async function?", "answer": "Executes asynchronously using Future."},
      ];
    } else if (widget.category == "Networking") {
      flashcards = [
        {"question": "What is an API?", "answer": "Application Programming Interface."},
        {"question": "What is HTTP?", "answer": "A protocol for web communication."},
        {"question": "What is REST?", "answer": "Representational State Transfer."},
      ];
    }
  }

  void nextCard(bool correct) {
    setState(() {
      if (correct) score++;
      if (currentIndex < flashcards.length - 1) {
        currentIndex++;
      } else {
        showDialog(
          context: context,
          builder: (context) => AlertDialog(
            title: Text("Quiz Completed"),
            content: Text("Your Score: $score/${flashcards.length}"),
            actions: [
              TextButton(
                onPressed: () {
                  Navigator.pop(context);
                  setState(() {
                    currentIndex = 0;
                    score = 0;
                  });
                },
                child: Text("Restart"),
              )
            ],
          ),
        );
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text("${widget.category} Flashcards")),
      body: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text("Score: $score", style: TextStyle(fontSize: 22, fontWeight: FontWeight.bold)),
            SizedBox(height: 20),
            Expanded(
              child: PageView.builder(
                itemCount: flashcards.length,
                controller: PageController(viewportFraction: 0.85),
                itemBuilder: (context, index) {
                  return flashcardCard(index);
                },
              ),
            ),
            SizedBox(height: 20),
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: [
                ElevatedButton(
                  onPressed: () => nextCard(false),
                  style: ElevatedButton.styleFrom(backgroundColor: Colors.red),
                  child: Text("Incorrect"),
                ),
                ElevatedButton(
                  onPressed: () => nextCard(true),
                  style: ElevatedButton.styleFrom(backgroundColor: Colors.green),
                  child: Text("Correct"),
                ),
              ],
            ),
            SizedBox(height: 20),
            ElevatedButton(
              onPressed: () async {
                final newFlashcard = await Navigator.push(
                  context,
                  MaterialPageRoute(builder: (context) => AddFlashcardScreen()),
                );
                if (newFlashcard != null) {
                  setState(() {
                    flashcards.add(newFlashcard);
                  });
                }
              },
              child: Text("Add Flashcard"),
            ),
          ],
        ),
      ),
    );
  }

  Widget flashcardCard(int index) {
    return Card(
      color: Colors.teal,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(15)),
      elevation: 8,
      child: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Center(
          child: Text(
            flashcards[index]["question"]!,
            style: TextStyle(fontSize: 22, fontWeight: FontWeight.bold, color: Colors.white),
            textAlign: TextAlign.center,
          ),
        ),
      ),
    );
  }
}

class AddFlashcardScreen extends StatelessWidget {
  final _questionController = TextEditingController();
  final _answerController = TextEditingController();

  void saveFlashcard(BuildContext context) {
    if (_questionController.text.isEmpty || _answerController.text.isEmpty) return;
    Navigator.pop(context, {"question": _questionController.text, "answer": _answerController.text});
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text("Add Flashcard")),
      body: Padding(
        padding: EdgeInsets.all(20),
        child: Column(children: [
          TextField(controller: _questionController, decoration: InputDecoration(labelText: "Question")),
          TextField(controller: _answerController, decoration: InputDecoration(labelText: "Answer")),
          SizedBox(height: 20),
          ElevatedButton(onPressed: () => saveFlashcard(context), child: Text("Save")),
        ]),
      ),
    );
  }
}
