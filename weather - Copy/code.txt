import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;

void main() => runApp(const WeatherApp());

class WeatherApp extends StatelessWidget {
  const WeatherApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Weather App',
      theme: ThemeData(primarySwatch: Colors.blue),
      home: const WeatherScreen(),
      debugShowCheckedModeBanner: false,
    );
  }
}

class WeatherScreen extends StatefulWidget {
  const WeatherScreen({super.key});

  @override
  State<WeatherScreen> createState() => _WeatherScreenState();
}

class _WeatherScreenState extends State<WeatherScreen> {
  final TextEditingController _controller = TextEditingController();
  String? temperature;
  String? description;
  String? error;
  bool isLoading = false;

  final String apiKey = 'b3c6f1102c2a2cd08562298da0e9d49b';

  Future<void> fetchWeather(String city) async {
    if (city.isEmpty) return;

    setState(() {
      isLoading = true;
      error = null;
      temperature = null;
      description = null;
    });

    final url = Uri.parse(
        'https://api.openweathermap.org/data/2.5/weather?q=$city&appid=$apiKey&units=metric');

    try {
      final response = await http.get(url);
      final data = json.decode(response.body);

      if (response.statusCode == 200) {
        setState(() {
          temperature = "${data['main']['temp']} °C";
          description = data['weather'][0]['description'].toString().toUpperCase();
        });
      } else {
        setState(() {
          error = data['message'] ?? 'Failed to get weather';
        });
      }
    } catch (e) {
      setState(() {
        error = 'Error: Could not fetch data';
      });
    } finally {
      setState(() {
        isLoading = false;
      });
    }
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text("Weather App")),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          children: [
            TextField(
              controller: _controller,
              decoration: const InputDecoration(
                labelText: 'Enter City',
                border: OutlineInputBorder(),
                prefixIcon: Icon(Icons.search),
              ),
            ),
            const SizedBox(height: 16),
            ElevatedButton(
              onPressed: () => fetchWeather(_controller.text.trim()),
              child: const Text('Get Weather'),
            ),
            const SizedBox(height: 24),
            if (isLoading) const CircularProgressIndicator(),
            if (error != null)
              Text(error!,
                  style: const TextStyle(fontSize: 18, color: Colors.red)),
            if (temperature != null && description != null)
              Column(
                children: [
                  Text("Temperature: $temperature",
                      style: const TextStyle(fontSize: 22)),
                  const SizedBox(height: 8),
                  Text("Condition: $description",
                      style: const TextStyle(fontSize: 20)),
                ],
              ),
          ],
        ),
      ),
    );
  }
}
